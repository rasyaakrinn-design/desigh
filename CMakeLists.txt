cmake_minimum_required(VERSION 3.10)
project(voicedsp)
add_library(voicedsp SHARED voice_dsp_android.cpp)
target_compile_features(voicedsp PRIVATE cxx_std_17)
set(LUA_HEADERS_DIR ${CMAKE_BINARY_DIR}/lua_headers)
file(MAKE_DIRECTORY ${LUA_HEADERS_DIR})
file(DOWNLOAD https://raw.githubusercontent.com/lua/lua/v5.4.7/src/lua.h ${LUA_HEADERS_DIR}/lua.h)
file(DOWNLOAD https://raw.githubusercontent.com/lua/lua/v5.4.7/src/lauxlib.h ${LUA_HEADERS_DIR}/lauxlib.h)
file(DOWNLOAD https://raw.githubusercontent.com/lua/lua/v5.4.7/src/luaconf.h ${LUA_HEADERS_DIR}/luaconf.h)
file(DOWNLOAD https://raw.githubusercontent.com/lua/lua/v5.4.7/src/lua.hpp ${LUA_HEADERS_DIR}/lua.hpp)
file(DOWNLOAD https://raw.githubusercontent.com/lua/lua/v5.4.7/src/lualib.h ${LUA_HEADERS_DIR}/lualib.h)
target_include_directories(voicedsp PRIVATE ${LUA_HEADERS_DIR})
